# creating a string of mRNA and calling it "test=mRNA"

#creating a function, setting a i to = 0, then creating a blank sequence
def get_amino_acids(mRNA):
    i = 0
    aa_sequence = []
    # creating a loop where i adds 3 and is asked if it is less than the mRNA sequence (24)
    while (i + 3) < len(mRNA):
    #creating a variable called codon that takes the first 3 letters of the sequence
        codon = mRNA[i:(i + 3)]
        # then creates a variable that plugs in the letters into the dictionary
        aa = genetic_code[codon]
        # if a stop codon is detected, it "breaks" or ends the loop
        if aa == "Stop":
            break
        # if a stop codon is not detected, it adds the output of the dictionary and puts it in the aa_sequence
        else:
            aa_sequence.append(aa)
        # advance to the next codon
        #adds 4 to i, moving it 4 places (so like from 0 to 4)
        #this is where the error is, because it should be 3 ( 3 DNA bases make up a codon )
        i = i + 4
        #returns the aa_sequence with all of the codons translated
    return "".join(aa_sequence)
